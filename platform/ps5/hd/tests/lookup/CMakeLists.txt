cmake_minimum_required(VERSION 3.20)
project(pe_hd_lookup_checks C)
option(PE_HD_LOOKUP_CHECKS "Build synthetic coordinator checks" OFF)
if(PE_PS5_HD_LOOKUP)
    if(PE_HD_LOOKUP_CHECKS AND NOT CMAKE_CROSSCOMPILING)
        find_package(SDL3 CONFIG REQUIRED)
        foreach(module image cache)
            add_library(pe-ps5-hd-${module} STATIC "${CMAKE_CURRENT_LIST_DIR}/../../pe_hd_${module}.c")
            target_include_directories(pe-ps5-hd-${module} PUBLIC "${CMAKE_CURRENT_LIST_DIR}/../..")
        endforeach()
        target_link_libraries(pe-ps5-hd-cache PUBLIC pe-ps5-hd-image SDL3::SDL3)
    endif()
    set(PE_PS5_HD_STAGING ON)
    set(PE_PS5_HD_GENERATION ON)
    include("${CMAKE_CURRENT_LIST_DIR}/../../staging.cmake")
    include("${CMAKE_CURRENT_LIST_DIR}/../../generation.cmake")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../lookup.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../../lookup.cmake")
if(PE_HD_LOOKUP_CHECKS)
    if(NOT TARGET pe-hd-lookup)
        message(FATAL_ERROR "Checks require PE_PS5_HD_LOOKUP=ON")
    endif()
    foreach(kind real faults)
        add_executable(check-lookup-${kind} "${CMAKE_CURRENT_LIST_DIR}/../check_lookup_${kind}.c")
        target_link_libraries(check-lookup-${kind} PRIVATE pe-hd-lookup)
        if(kind STREQUAL "real")
            target_link_libraries(check-lookup-${kind} PRIVATE pe-hd-vram)
        endif()
        set_target_properties(check-lookup-${kind} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        if(MSVC)
            target_compile_options(check-lookup-${kind} PRIVATE /UNDEBUG /W4)
        else()
            target_compile_options(check-lookup-${kind} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        endif()
        if(NOT CMAKE_CROSSCOMPILING)
            enable_testing()
            add_test(NAME hd-lookup-${kind} COMMAND check-lookup-${kind})
            set_tests_properties(hd-lookup-${kind} PROPERTIES TIMEOUT 30)
        endif()
    endforeach()
elseif(NOT PE_PS5_HD_LOOKUP AND (TARGET pe-hd-lookup OR TARGET pe-hd-pipeline OR TARGET SDL3::SDL3))
    message(FATAL_ERROR "Default OFF created dependency targets")
endif()
