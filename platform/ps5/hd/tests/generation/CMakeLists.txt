cmake_minimum_required(VERSION 3.20)
project(pe_hd_generation_checks C)
include("${CMAKE_CURRENT_LIST_DIR}/../../generation.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../../generation.cmake")
if(PE_PS5_HD_GENERATION)
    add_executable(check-generation "${CMAKE_CURRENT_LIST_DIR}/../check_generation.c")
    target_link_libraries(check-generation PRIVATE pe-hd-generation)
    set_target_properties(check-generation PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(check-generation PRIVATE /UNDEBUG /W4)
    else()
        target_compile_options(check-generation PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        target_compile_options(pe-hd-generation PRIVATE -Wall -Wextra -Werror)
    endif()
    if(NOT CMAKE_CROSSCOMPILING)
        enable_testing()
        add_test(NAME hd-generation COMMAND check-generation)
        set_tests_properties(hd-generation PROPERTIES TIMEOUT 30)
    endif()
elseif(TARGET pe-hd-generation)
    message(FATAL_ERROR "Generation target present with default OFF")
endif()
