cmake_minimum_required(VERSION 3.20)
project(pe_hd_lookup_snapshot_checks C)
# Reuse the developer harness without changing production staging defaults.
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." staging-checks)
if(NOT TARGET check-lookup-real OR NOT TARGET pe-hd-snapshot)
    message(FATAL_ERROR "Requires staging checks and snapshot/identity/generation options")
endif()
add_executable(check-snapshot "${CMAKE_CURRENT_LIST_DIR}/../check_snapshot.c")
target_link_libraries(check-snapshot PRIVATE pe-hd-snapshot pe-hd-vram)
if(MSVC)
    target_compile_options(check-snapshot PRIVATE /UNDEBUG /W4 /WX)
else()
    target_compile_options(check-snapshot PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()
if(NOT CMAKE_CROSSCOMPILING)
    enable_testing()
    add_test(NAME snapshot COMMAND check-snapshot)
endif()
