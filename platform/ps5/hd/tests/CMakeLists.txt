cmake_minimum_required(VERSION 3.20)
project(pe_hd_developer_checks C)
option(PE_HD_CHECKS "Build synthetic checks (host run; cross compile only)" OFF)
# Explicit test-only host path. Production staging retains its public SDK guard.
if(PE_HD_CHECKS AND NOT CMAKE_CROSSCOMPILING)
    find_package(SDL3 CONFIG REQUIRED)
    foreach(module image cache)
        add_library(pe-ps5-hd-${module} STATIC "${CMAKE_CURRENT_LIST_DIR}/../pe_hd_${module}.c")
        target_include_directories(pe-ps5-hd-${module} PUBLIC "${CMAKE_CURRENT_LIST_DIR}/..")
        set_target_properties(pe-ps5-hd-${module} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    endforeach()
    target_link_libraries(pe-ps5-hd-cache PUBLIC pe-ps5-hd-image SDL3::SDL3)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../staging.cmake")
# A second include must be safe and must not duplicate targets.
include("${CMAKE_CURRENT_LIST_DIR}/../staging.cmake")
if(NOT PE_PS5_HD_STAGING)
    foreach(unexpected pe-hd-staging pe-hd-filename pe-ps5-hd-image pe-ps5-hd-cache pe-hd-index pe-hd-pipeline pe-hd-vram pe-hd-generation pe-hd-lookup pe-hd-pack pe-hd-manifest SDL3::SDL3)
        if(TARGET ${unexpected})
            message(FATAL_ERROR "Default OFF unexpectedly creates target ${unexpected}")
        endif()
    endforeach()
endif()
if(PE_HD_CHECKS)
    if(NOT TARGET pe-hd-staging)
        message(FATAL_ERROR "PE_HD_CHECKS requires PE_PS5_HD_STAGING=ON")
    endif()
    enable_testing()
    # The real lookup fixture consumes snapshot-produced full-page identities.
    # These options are local to the synthetic checks, never production defaults.
    set(PE_PS5_HD_IDENTITY ON)
    set(PE_PS5_HD_SNAPSHOT ON)
    include("${CMAKE_CURRENT_LIST_DIR}/../snapshot.cmake")
    foreach(module image cache index pipeline vram generation lookup-real lookup-faults pack manifest)
        if(module MATCHES "^lookup-")
            string(REPLACE "lookup-" "lookup_" check_name "${module}")
            set(source "${CMAKE_CURRENT_LIST_DIR}/check_${check_name}.c")
        elseif(module STREQUAL "index" OR module STREQUAL "pipeline" OR module STREQUAL "generation" OR module STREQUAL "pack" OR module STREQUAL "manifest")
            set(source "${CMAKE_CURRENT_LIST_DIR}/check_${module}.c")
        else()
            set(source "${CMAKE_CURRENT_LIST_DIR}/../check_${module}.c")
        endif()
        add_executable(check-${module} "${source}")
        target_include_directories(check-${module} PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
        target_link_libraries(check-${module} PRIVATE pe-hd-staging)
        if(module STREQUAL "lookup-real")
            target_link_libraries(check-${module} PRIVATE pe-hd-snapshot)
        endif()
        set_target_properties(check-${module} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        # Assertions execute test operations; keep them enabled in Release too.
        if(MSVC)
            target_compile_options(check-${module} PRIVATE /UNDEBUG)
        else()
            target_compile_options(check-${module} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        endif()
        if(NOT CMAKE_CROSSCOMPILING)
            add_test(NAME hd-${module} COMMAND check-${module})
            set_tests_properties(hd-${module} PROPERTIES TIMEOUT 30)
        endif()
    endforeach()
endif()
