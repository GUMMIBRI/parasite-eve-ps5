cmake_minimum_required(VERSION 3.20)
project(pe_hd_end_to_end_checks C)
# No root/staging production changes. Explicit opt-in developer harness only.
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." staging-checks)
if(NOT TARGET pe-hd-snapshot OR NOT TARGET pe-hd-staging)
    message(FATAL_ERROR "Requires PE_HD_CHECKS=ON and PE_PS5_HD_STAGING=ON")
endif()
add_executable(check-end-to-end ../check_end_to_end.c)
target_include_directories(check-end-to-end PRIVATE ..)
target_link_libraries(check-end-to-end PRIVATE pe-hd-staging pe-hd-snapshot)
set_target_properties(check-end-to-end PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
    target_compile_options(check-end-to-end PRIVATE /UNDEBUG /W4 /WX)
else()
    target_compile_options(check-end-to-end PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()
if(NOT CMAKE_CROSSCOMPILING)
    enable_testing()
    add_test(NAME hd-end-to-end COMMAND check-end-to-end)
    set_tests_properties(hd-end-to-end PROPERTIES TIMEOUT 15)
endif()
