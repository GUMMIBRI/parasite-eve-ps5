cmake_minimum_required(VERSION 3.20)
project(pe_hd_snapshot_check C)
include("${CMAKE_CURRENT_LIST_DIR}/../../snapshot.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../../snapshot.cmake")
if(TARGET pe-hd-snapshot)
    include("${CMAKE_CURRENT_LIST_DIR}/../../vram.cmake")
    if(NOT TARGET pe-hd-vram)
        message(FATAL_ERROR "Fixture requires PE_PS5_HD_VRAM")
    endif()
    add_executable(check-snapshot "${CMAKE_CURRENT_LIST_DIR}/../check_snapshot.c")
    target_link_libraries(check-snapshot PRIVATE pe-hd-snapshot pe-hd-vram)
    foreach(t pe-hd-snapshot pe-hd-identity pe-hd-generation pe-hd-vram check-snapshot)
        if(MSVC)
            target_compile_options(${t} PRIVATE /UNDEBUG /W4 /WX)
        else()
            target_compile_options(${t} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        endif()
    endforeach()
    if(NOT CMAKE_CROSSCOMPILING)
        enable_testing()
        add_test(NAME snapshot COMMAND check-snapshot)
        set_tests_properties(snapshot PROPERTIES TIMEOUT 15)
    endif()
else()
    get_property(targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    if(targets)
        message(FATAL_ERROR "Default OFF created targets: ${targets}")
    endif()
endif()
