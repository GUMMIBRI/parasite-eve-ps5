cmake_minimum_required(VERSION 3.20)
project(pe_hd_identity_check C)
include("${CMAKE_CURRENT_LIST_DIR}/../../identity.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../../identity.cmake")
if(TARGET pe-hd-identity)
    add_executable(check-identity "${CMAKE_CURRENT_LIST_DIR}/../check_identity.c")
    target_link_libraries(check-identity PRIVATE pe-hd-identity)
    if(MSVC)
        target_compile_options(check-identity PRIVATE /UNDEBUG /W4 /WX)
    else()
        target_compile_options(check-identity PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        target_compile_options(pe-hd-identity PRIVATE -Wall -Wextra -Werror)
    endif()
    if(NOT CMAKE_CROSSCOMPILING)
        enable_testing()
        add_test(NAME identity COMMAND check-identity)
        set_tests_properties(identity PROPERTIES TIMEOUT 15)
    endif()
else()
    get_property(targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    if(targets)
        message(FATAL_ERROR "Default OFF created unexpected targets: ${targets}")
    endif()
endif()
